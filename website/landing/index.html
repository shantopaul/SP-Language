class Greeter<T> {
    public string greeting;
    
    public constructor(greeting: string) {
        this.greeting = greeting;
    }

    public fn welcome(user: T) {
        println("${this.greeting}, ${user}!");
    }
}

fn main() -> int {
    let app = new Greeter<string>("Hello");
    app.welcome("SP 1.0 World");
    return 0;
}
